4. STEP 3: TRIGGERING PUSH NOTIFICATIONS
================================================================================

Push notifications are triggered in two ways:

--- METHOD A: AUTOMATIC SYSTEM TRIGGERS (BUILT-IN) ---
You don't need to do anything manual. The backend automatically dispatches push
notifications when:
  1. PDF Transfer: When an admin uploads and assigns a COA/TDS document to a
     distributor company, the backend looks up all active users in that company,
     checks their notification preferences, and sends a push alert:
       Title: "New Document Received"
       Body:  "Batch_COA_1029.pdf has been shared with your company."
       Data:  { "type": "PDF", "pdfId": "..." }

  2. News Publish: When an industry news article is published:
       Title: "New Industry Update"
       Body:  "A new article has been published: [Article Title]"
       Data:  { "type": "NEWS", "newsId": "..." }

--- METHOD B: EXTERNAL SERVER / API TRIGGER ---
If you have an external ERP, automated scanner, or script that wants to trigger
a notification for distributor(s) when a file is uploaded:

ENDPOINT:
  POST /api/notifications/send

HEADERS:
  Content-Type: application/json

REQUEST BODY:
{
  "distributorIds": [
    "dist_clx111111111",
    "dist_clx222222222"
  ],
  "pdfId": "pdf_clx999999999",
  "apiKey": "your-secret-api-key"
}

SUCCESS RESPONSE (200 OK):
{
  "success": true,
  "message": "Notifications sent to 2 distributor(s)",
  "data": {
    "notificationCount": 2,
    "pdf": {
      "id": "pdf_clx999999999",
      "fileName": "Certificate_Of_Analysis_Ethanol_99.pdf"
    }
  }
}

B) REACT NATIVE / EXPO (TYPESCRIPT)

```typescript
import * as Notifications from 'expo-notifications';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { Platform } from 'react-native';

const API_BASE = 'https://your-domain.com/api';

// 1. Authenticate and save token
export async function loginMobile(email: string, password: string) {
  const res = await fetch(`${API_BASE}/auth/login`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ email, password }),
  });
  const json = await res.json();
  if (json.success) {
    await AsyncStorage.setItem('authToken', json.data.token);
    // After login, register push token
    await registerForPushNotifications();
  }
  return json;
}

// 2. Obtain device push token & register with Spentica backend
export async function registerForPushNotifications() {
  const { status: existingStatus } = await Notifications.getPermissionsAsync();
  let finalStatus = existingStatus;

  if (existingStatus !== 'granted') {
    const { status } = await Notifications.requestPermissionsAsync();
    finalStatus = status;
  }

  if (finalStatus !== 'granted') {
    console.warn('Push notification permission denied');
    return;
  }

  const tokenData = await Notifications.getDevicePushTokenAsync();
  const pushToken = tokenData.data;
  const authToken = await AsyncStorage.getItem('authToken');

  await fetch(`${API_BASE}/devices`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      Authorization: `Bearer ${authToken}`,
    },
    body: JSON.stringify({
      token: pushToken,
      platform: Platform.OS === 'ios' ? 'IOS' : 'ANDROID',
      deviceInfo: {
        model: Platform.constants.Model || 'MobileDevice',
        osVersion: String(Platform.Version),
        appVersion: '1.0.0',
      },
    }),
  });
}

// 3. Fetch notifications feed
export async function fetchNotifications(unreadOnly = false) {
  const authToken = await AsyncStorage.getItem('authToken');
  const url = `${API_BASE}/notifications?limit=20${unreadOnly ? '&readFlag=false' : ''}`;

  const res = await fetch(url, {
    headers: { Authorization: `Bearer ${authToken}` },
  });
  return await res.json();
}

// 4. Mark notifications as read
export async function markAsRead(notificationIds: string[]) {
  const authToken = await AsyncStorage.getItem('authToken');
  await fetch(`${API_BASE}/notifications/mark-read`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      Authorization: `Bearer ${authToken}`,
    },
    body: JSON.stringify({ notificationIds }),
  });
}
```

C) FLUTTER (DART)
--------------------------------------------------------------------------------

```dart
import 'dart:convert';
import 'dart:io';
import 'package:http/http.dart' as http;
import 'package:firebase_messaging/firebase_messaging.dart';
import 'package:flutter_secure_storage/flutter_secure_storage.dart';

class SpenticaNotificationService {
  static const String baseUrl = 'https://your-domain.com/api';
  final _storage = const FlutterSecureStorage();
  final FirebaseMessaging _fcm = FirebaseMessaging.instance;

  // 1. Login User
  Future<bool> login(String email, String password) async {
    final response = await http.post(
      Uri.parse('$baseUrl/auth/login'),
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode({'email': email, 'password': password}),
    );

    if (response.statusCode == 200) {
      final data = jsonDecode(response.body);
      final token = data['data']['token'];
      await _storage.write(key: 'jwt_token', value: token);
      await registerDevice();
      return true;
    }
    return false;
  }

  // 2. Register Device Push Token
  Future<void> registerDevice() async {
    NotificationSettings settings = await _fcm.requestPermission();
    if (settings.authorizationStatus != AuthorizationStatus.authorized) return;

    String? fcmToken = await _fcm.getToken();
    String? jwt = await _storage.read(key: 'jwt_token');

    if (fcmToken != null && jwt != null) {
      await http.post(
        Uri.parse('$baseUrl/devices'),
        headers: {
          'Content-Type': 'application/json',
          'Authorization': 'Bearer $jwt',
        },
        body: jsonEncode({
          'token': fcmToken,
          'platform': Platform.isIOS ? 'IOS' : 'ANDROID',
          'deviceInfo': {
            'osVersion': Platform.operatingSystemVersion,
            'appVersion': '1.0.0',
          },
        }),
      );
    }
  }

  // 3. Fetch Notification List & Unread Count
  Future<Map<String, dynamic>> getNotifications({bool unreadOnly = false}) async {
    String? jwt = await _storage.read(key: 'jwt_token');
    final query = unreadOnly ? '?readFlag=false' : '';

    final res = await http.get(
      Uri.parse('$baseUrl/notifications$query'),
      headers: {'Authorization': 'Bearer $jwt'},
    );
    return jsonDecode(res.body);
  }

  // 4. Mark Notification Read
  Future<void> markRead(List<String> ids) async {
    String? jwt = await _storage.read(key: 'jwt_token');
    await http.post(
      Uri.parse('$baseUrl/notifications/mark-read'),
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer $jwt',
      },
      body: jsonEncode({'notificationIds': ids}),
    );
  }
}
```

10. PRODUCTION SETUP CHECKLIST (FIREBASE FCM & APNs)
================================================================================

To ensure physical mobile devices receive lock-screen push alerts:

1. FIREBASE SETUP:
   - Create a project in Google Firebase Console (https://console.firebase.google.com).
   - Add your Android app (download `google-services.json`) and iOS app
     (download `GoogleService-Info.plist`).
   - Go to Project Settings -> Service Accounts -> Generate New Private Key.
   - Save the JSON credential and set environment variable in `.env`:
     FIREBASE_SERVICE_ACCOUNT_KEY='{"type":"service_account",...}'

2. APPLE APNs (iOS):
   - In Apple Developer Portal, create an APNs Key (.p8 file).
   - Upload the APNs Key into Firebase Console under Project Settings -> Cloud Messaging -> Apple app configuration.

3. SPENTICA BACKEND DISPATCH:
   - The Spentica backend already contains `lib/services/notification-service.ts`
     with `sendPushNotification(userId, payload)`.
   - When a Firebase Admin SDK instance is initialized with `FIREBASE_SERVICE_ACCOUNT_KEY`,
     the backend sends notifications via `firebase-admin.messaging().sendEachForMulticast()`
     to all active tokens stored in the `device_tokens` database table.
